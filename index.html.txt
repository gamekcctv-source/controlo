    function getLuandaDateParts() {
      const now = new Date();
      try {
        const timeStr = new Intl.DateTimeFormat("pt-PT", {
          timeZone: "Africa/Luanda",
          hour: "2-digit",
          minute: "2-digit",
          second: "2-digit",
          hour12: false
        }).format(now);

        const dateStr = new Intl.DateTimeFormat("pt-PT", {
          timeZone: "Africa/Luanda",
          day: "2-digit",
          month: "2-digit",
          year: "numeric"
        }).format(now);

        const hmStr = new Intl.DateTimeFormat("pt-PT", {
          timeZone: "Africa/Luanda",
          hour: "2-digit",
          minute: "2-digit",
          hour12: false
        }).format(now);

        return { timeStr, dateStr, hmStr };
      } catch (e) {
        const pad = (n) => String(n).padStart(2, "0");
        return {
          timeStr: `${pad(now.getHours())}:${pad(now.getMinutes())}:${pad(now.getSeconds())}`,
          dateStr: `${pad(now.getDate())}/${pad(now.getMonth() + 1)}/${now.getFullYear()}`,
          hmStr: `${pad(now.getHours())}:${pad(now.getMinutes())}`
        };
      }
    }

    // =========================================================================
    // 5. TOUCH SIGNATURE PAD (CANVAS)
    // =========================================================================
    let sigCanvas, sigCtx, isDrawing = false;

    function initSignaturePad() {
      sigCanvas = document.getElementById("signature-pad");
      sigCtx = sigCanvas.getContext("2d");

      const resizeCanvas = () => {
        const rect = sigCanvas.getBoundingClientRect();
        const dpr = window.devicePixelRatio || 1;
        sigCanvas.width = rect.width * dpr;
        sigCanvas.height = rect.height * dpr;
        sigCtx.scale(dpr, dpr);
        sigCtx.lineWidth = 2.2;
        sigCtx.lineCap = "round";
        sigCtx.lineJoin = "round";
        sigCtx.strokeStyle = "#0f172a";
      };

      resizeCanvas();
      window.addEventListener("resize", () => {
        if (!hasSignatureDrawn) resizeCanvas();
      });

      const getCoords = (e) => {
        const rect = sigCanvas.getBoundingClientRect();
        const clientX = e.touches ? e.touches[0].clientX : e.clientX;
        const clientY = e.touches ? e.touches[0].clientY : e.clientY;
        return { x: clientX - rect.left, y: clientY - rect.top };
      };

      const startDraw = (e) => {
        isDrawing = true;
        hasSignatureDrawn = true;
        document.getElementById("sig-placeholder").classList.add("hidden");
        const { x, y } = getCoords(e);
        sigCtx.beginPath();
        sigCtx.moveTo(x, y);
      };

      const draw = (e) => {
        if (!isDrawing) return;
        if (e.cancelable) e.preventDefault();
        const { x, y } = getCoords(e);
        sigCtx.lineTo(x, y);
        sigCtx.stroke();
      };

      const stopDraw = () => {
        if (isDrawing) {
          isDrawing = false;
          sigCtx.closePath();
        }
      };

      sigCanvas.addEventListener("mousedown", startDraw);
      sigCanvas.addEventListener("mousemove", draw);
      window.addEventListener("mouseup", stopDraw);

      sigCanvas.addEventListener("touchstart", startDraw, { passive: false });
      sigCanvas.addEventListener("touchmove", draw, { passive: false });
      sigCanvas.addEventListener("touchend", stopDraw);
    }

    function clearSignaturePad() {
      const rect = sigCanvas.getBoundingClientRect();
      sigCtx.clearRect(0, 0, rect.width, rect.height);
      hasSignatureDrawn = false;
      document.getElementById("sig-placeholder").classList.remove("hidden");
    }

    function getSignatureDataURL() {
      if (!hasSignatureDrawn) return null;
      return sigCanvas.toDataURL("image/png");
    }

    function handleFormSubmit(e) {
      e.preventDefault();

      // Resolve collaborator from either search or select
      if (!selectedCollaborator) {
        const typed = normalizeText(document.getElementById("collab-search").value.trim());
        const exact = STAFF_LIST.find((p) => normalizeText(p.name) === typed);
        if (exact) {
          selectedCollaborator = exact;
        } else {
          showToast("Por favor, selecione um colaborador válido da lista.", "error");
          document.getElementById("collab-search").focus();
          return;
        }
      }

      const editId = document.getElementById("edit-record-id").value;
      const time = document.getElementById("purchase-time").value;
      const items = document.getElementById("purchase-items").value.trim();
      const receipt = document.getElementById("receipt-number").value.trim();
      const amount = parseFloat(document.getElementById("total-kz").value);
      const cashier = document.getElementById("cashier-select").value;
      const status = document.getElementById("initial-status").value;
      const cctvOperator = document.getElementById("cctv-operator-select").value;
      const shift = document.getElementById("shift-select").value;
      const quickConfirm = document.getElementById("quick-confirm-check").checked;
      const signatureData = getSignatureDataURL();
      const { dateStr } = getLuandaDateParts();

      if (isNaN(amount) || amount < 0) {
        showToast("Insira um valor válido em Kwanzas (Kz).", "error");
        return;
      }

      if (editId) {
        const idx = purchases.findIndex((p) => p.id === editId);
        if (idx !== -1) {
          purchases[idx] = {
            ...purchases[idx],
            collaboratorName: selectedCollaborator.name,
            collaboratorRole: selectedCollaborator.role,
            time,
            items,
            receipt,
            amount,
            cashier,
            status,
            cctvOperator,
            signature: signatureData || purchases[idx].signature,
            quickConfirm: signatureData ? false : quickConfirm
          };
          showToast(`Registo de ${selectedCollaborator.name} atualizado!`, "success");
        }
      } else {
        const newRecord = {
          id: "REC-" + Date.now() + "-" + Math.floor(Math.random() * 1000),
          date: dateStr,
          time,
          collaboratorName: selectedCollaborator.name,
          collaboratorRole: selectedCollaborator.role,
          items,
          receipt,
          amount,
          cashier,
          status,
          cctvOperator,
          shift,
          signature: signatureData,
          quickConfirm: !signatureData && quickConfirm
        };
        purchases.unshift(newRecord);
        showToast(`Compra de ${selectedCollaborator.name} registada!`, "success");
      }

      savePurchases();
      resetForm();
      renderAll();
    }

    function resetForm() {
      document.getElementById("purchase-form").reset();
      document.getElementById("edit-record-id").value = "";
      document.getElementById("form-heading").textContent = "Novo Registo de Compra";
      document.getElementById("submit-btn-text").textContent = "Registar Compra na Folha";
      document.getElementById("cancel-edit-btn").classList.add("hidden");
      clearSelectedCollaborator();

    function editRecord(id) {
      const rec = purchases.find((p) => p.id === id);
      if (!rec) return;

      const person = STAFF_LIST.find((p) => p.name === rec.collaboratorName) || {
        name: rec.collaboratorName,
        role: rec.collaboratorRole
      };
      selectCollaborator(person);

      document.getElementById("edit-record-id").value = rec.id;
      document.getElementById("purchase-time").value = rec.time;
      document.getElementById("purchase-items").value = rec.items;
      document.getElementById("receipt-number").value = rec.receipt;
      document.getElementById("total-kz").value = rec.amount;
      document.getElementById("cashier-select").value = rec.cashier;
      document.getElementById("initial-status").value = rec.status;
      document.getElementById("quick-confirm-check").checked = !!rec.quickConfirm;

      document.getElementById("form-heading").textContent = "Editar Registo de Compra";
      document.getElementById("submit-btn-text").textContent = "Guardar Alterações";
      document.getElementById("cancel-edit-btn").classList.remove("hidden");

      switchTab("registo");
      window.scrollTo({ top: 0, behavior: "smooth" });
    }

    function setRecordStatus(id, newStatus) {
      const rec = purchases.find((p) => p.id === id);
      if (!rec) return;
      rec.status = newStatus;
      rec.cctvOperator = document.getElementById("cctv-operator-select").value;
      savePurchases();
      renderAll();

      const label =
        newStatus === "Conferido OK"
          ? "Conferido OK pelo CCTV"
          : newStatus === "Divergência"
          ? "Marcado com Divergência!"
          : "Marcado como Pendente";
      showToast(`${rec.collaboratorName}: ${label}`, newStatus === "Divergência" ? "error" : "success");
    }

    function deleteRecord(id) {
      const rec = purchases.find((p) => p.id === id);
      if (!rec) return;

      openConfirmModal(
        "Eliminar Registo",
        `Tem a certeza que pretende eliminar o registo de ${rec.collaboratorName} (Talão: ${rec.receipt} — ${formatKz(rec.amount)})?`,
        () => {
          purchases = purchases.filter((p) => p.id !== id);
          savePurchases();
          renderAll();
          showToast("Registo eliminado da folha.", "info");
        }
      );
    }

    function confirmClearDay() {
      if (purchases.length === 0) {
        showToast("A folha já se encontra limpa.", "info");
        return;
      }
      openConfirmModal(
        "Limpar Dia / Iniciar Novo Turno",
        `Esta ação vai apagar todos os ${purchases.length} registos atuais para iniciar um novo dia/turno. Recomenda-se exportar CSV ou copiar o resumo WhatsApp antes. Deseja continuar?`,
        () => {
          purchases = [];
          savePurchases();
          resetForm();
          renderAll();
          showToast("Folha limpa! Pronto para o novo turno.", "success");
        }
      );
    }

    function formatKz(value) {
      const num = Number(value) || 0;
      return (
        num.toLocaleString("pt-PT", {
          minimumFractionDigits: 2,
          maximumFractionDigits: 2
        }) + " Kz"
      );
    }

    function renderAll() {
      updateKPIs();
      renderPurchases();
      renderTeamView();
    }

    function updateKPIs() {
      const totalCount = purchases.length;
      const totalAmount = purchases.reduce((sum, r) => sum + (Number(r.amount) || 0), 0);
      const okCount = purchases.filter((r) => r.status === "Conferido OK").length;
      const pendingCount = purchases.filter((r) => r.status === "Pendente CCTV").length;
      const divCount = purchases.filter((r) => r.status === "Divergência").length;

      document.getElementById("kpi-count").textContent = totalCount;
      document.getElementById("kpi-total-kz").textContent = formatKz(totalAmount);
      document.getElementById("kpi-ok").textContent = okCount;
      document.getElementById("kpi-pending").textContent = pendingCount;
      document.getElementById("kpi-div").textContent = divCount;

      document.getElementById("count-all").textContent = totalCount;
      document.getElementById("count-ok").textContent = okCount;
      document.getElementById("count-pending").textContent = pendingCount;
      document.getElementById("count-div").textContent = divCount;

      document.getElementById("tfoot-count").textContent = totalCount;
      document.getElementById("tfoot-total").textContent = formatKz(totalAmount);
      document.getElementById("tfoot-summary").textContent = `OK: ${okCount} | Pendentes: ${pendingCount} | Divergências: ${divCount}`;
    }

          btn.classList.remove("ring-2", "ring-offset-1", "ring-emerald-500");

      renderPurchases();
    }

    function renderTeamView() {
      const roleFilter = document.getElementById("team-role-filter").value;
      const onlyActive = document.getElementById("team-only-active").checked;
      const grid = document.getElementById("team-grid");

      const statsByCollab = {};
      purchases.forEach((p) => {
        if (!statsByCollab[p.collaboratorName]) {
          statsByCollab[p.collaboratorName] = { count: 0, total: 0 };
        }
        statsByCollab[p.collaboratorName].count += 1;
        statsByCollab[p.collaboratorName].total += Number(p.amount) || 0;
      });

      const list = STAFF_LIST.filter((person) => {
        const matchesRole = roleFilter === "ALL" || person.role === roleFilter;
        const stat = statsByCollab[person.name] || { count: 0, total: 0 };
        const matchesActive = !onlyActive || stat.count > 0;
        return matchesRole && matchesActive;
      });

      if (list.length === 0) {
        grid.innerHTML = `<div class="col-span-full py-10 text-center text-sm text-slate-500">Nenhum colaborador encontrado para este filtro.</div>`;
        return;
      }

      grid.innerHTML = list
        .map((person) => {
          const stat = statsByCollab[person.name] || { count: 0, total: 0 };
          const hasPurchases = stat.count > 0;
          return `
            <div class="p-3.5 rounded-2xl border ${
              hasPurchases
                ? "border-emerald-500/40 bg-emerald-500/5 dark:bg-emerald-500/10"
                : "border-slate-200 dark:border-slate-800 bg-slate-50/50 dark:bg-slate-800/40"
            } flex items-center justify-between gap-3">
              <div class="min-w-0">
                <div class="font-bold text-sm text-slate-900 dark:text-white truncate">${escapeHTML(person.name)}</div>
                <div class="text-xs font-medium text-slate-500 dark:text-slate-400">${escapeHTML(person.role)}</div>
              </div>
              <div class="text-right shrink-0">
                <div class="font-mono text-sm font-extrabold ${
                  hasPurchases ? "text-emerald-600 dark:text-emerald-400" : "text-slate-400"
                }">
                  ${formatKz(stat.total)}
                </div>
                <div class="text-[11px] font-semibold text-slate-500 dark:text-slate-400">
                  ${stat.count} ${stat.count === 1 ? "compra" : "compras"}
                </div>
              </div>
            </div>
          `;
        })
        .join("");
    }

    function copyWhatsAppSummary() {
      const { dateStr, timeStr } = getLuandaDateParts();
      const cctvOp = document.getElementById("cctv-operator-select").value;
      const shift = document.getElementById("shift-select").value;

      const totalCount = purchases.length;
      const totalAmount = purchases.reduce((s, r) => s + (Number(r.amount) || 0), 0);
      const okCount = purchases.filter((r) => r.status === "Conferido OK").length;
      const pendingCount = purchases.filter((r) => r.status === "Pendente CCTV").length;
      const divCount = purchases.filter((r) => r.status === "Divergência").length;

      let lines = [
        `*CONTROLO DE COMPRAS COLABORADORES — GAMEK FRESMART EXPRESS 1*`,
        `📅 *Data:* ${dateStr} (${timeStr})`,
        `🕒 *Turno:* ${shift}`,
        `📹 *Operador CCTV:* ${cctvOp}`,
        `------------------------------------`,
        `📊 *RESUMO DO TURNO:*`,
        `• Total Registos: *${totalCount}*`,
        `• Valor Total: *${formatKz(totalAmount)}*`,
        `• 🟢 Conferidos OK: *${okCount}*`,
        `• 🟡 Pendentes CCTV: *${pendingCount}*`,
        `• 🔴 Divergências: *${divCount}*`,
        `------------------------------------`
      ];

      if (purchases.length === 0) {
        lines.push(`_Sem compras registadas até ao momento._`);
      } else {
        lines.push(`*DETALHE DAS COMPRAS:*`);
        purchases
          .slice()
          .reverse()
          .forEach((r, i) => {
            const icon = r.status === "Conferido OK" ? "🟢" : r.status === "Divergência" ? "🔴" : "🟡";
            lines.push(
              `${i + 1}. ${icon} *[${r.time}] ${r.collaboratorName}* (${r.collaboratorRole})\n   Artigos: ${r.items}\n   Talão: ${r.receipt} | Cx: ${r.cashier} | *${formatKz(r.amount)}*`
            );
          });
      }

      const textToCopy = lines.join("\n");

      // Reliable clipboard copy inside sandboxed/iframe environments
      const textArea = document.createElement("textarea");
      textArea.value = textToCopy;
      textArea.style.position = "fixed";
      textArea.style.left = "-9999px";
      document.body.appendChild(textArea);
      textArea.select();
      try {
        document.execCommand("copy");
        showToast("Resumo copiado! Pronto para colar no WhatsApp.", "success");
      } catch (err) {
        showToast("Não foi possível copiar automaticamente.", "error");
      }
      document.body.removeChild(textArea);
    }

    function exportCSV() {
      if (purchases.length === 0) {
        showToast("Não existem registos para exportar.", "error");
        return;
      }

      const { dateStr } = getLuandaDateParts();
      const headers = [
        "Data",
        "Hora",
        "Nome do Colaborador",
        "Funcao / Cargo",
        "Artigos Comprados",
        "N. Talao / Fatura",
        "Valor Total (Kz)",
        "Operador de Caixa",
        "Estado CCTV",
        "Operador CCTV",
        "Turno",
        "Assinatura"
      ];

      const escapeCSV = (val) => `"${String(val ?? "").replace(/"/g, '""')}"`;

      const rows = purchases
        .slice()
        .reverse()
        .map((r) => [
          r.date || dateStr,
          r.time,
          r.collaboratorName,
          r.collaboratorRole,
          r.items,
          r.receipt,
          Number(r.amount).toFixed(2).replace(".", ","),
          r.cashier,
          r.status,
          r.cctvOperator,
          r.shift,
          r.signature ? "Assinado Digitalmente" : r.quickConfirm ? "Confirmado Presencial" : "Pendente"
        ]);

      // UTF-8 BOM + Semicolon separator for Portuguese Excel compatibility
      const csvContent =
        "\uFEFF" +
        [headers.map(escapeCSV).join(";"), ...rows.map((row) => row.map(escapeCSV).join(";"))].join("\r\n");

      const blob = new Blob([csvContent], { type: "text/csv;charset=utf-8;" });
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      const cleanDate = dateStr.replace(/\//g, "-");
      a.href = url;
      a.download = `Controlo_Compras_Gamek_Fresmart_Express1_${cleanDate}.csv`;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(url);

      showToast("Ficheiro CSV / Excel transferido com sucesso!", "success");
    }

    function openConfirmModal(title, message, onConfirm) {
      const modal = document.getElementById("confirm-modal");
      document.getElementById("confirm-title").textContent = title;
      document.getElementById("confirm-message").textContent = message;

      const okBtn = document.getElementById("confirm-ok-btn");
      const cancelBtn = document.getElementById("confirm-cancel-btn");

      const cleanup = () => {
        modal.classList.add("hidden");
        modal.classList.remove("flex");
        okBtn.onclick = null;
        cancelBtn.onclick = null;
      };

      okBtn.onclick = () => {
        cleanup();
        onConfirm();
      };
      cancelBtn.onclick = cleanup;

      modal.classList.remove("hidden");
      modal.classList.add("flex");
    }

    function showToast(message, type = "info") {
      const container = document.getElementById("toast-container");
      const toast = document.createElement("div");

      const styles =
        type === "success"
          ? "bg-emerald-600 text-white border-emerald-500"
          : type === "error"
          ? "bg-rose-600 text-white border-rose-500"
          : "bg-slate-800 text-white border-slate-700";

      toast.className = `pointer-events-auto px-4 py-3 rounded-xl border shadow-xl text-xs sm:text-sm font-bold flex items-center justify-between gap-3 transition-all duration-300 ${styles}`;
      toast.innerHTML = `<span>${escapeHTML(message)}</span>`;

      container.appendChild(toast);
      setTimeout(() => {
        toast.style.opacity = "0";
        toast.style.transform = "translateY(-8px)";
        setTimeout(() => toast.remove(), 300);
      }, 3200);
    }

    function escapeHTML(str) {
      return String(str ?? "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;");
    }
